def read_excel_with_multiple_sheets(file_path):
    """
    读取包含多个sheet的Excel文件
    
    参数:
        file_path (str): Excel文件路径
        
    返回:
        dict: 以sheet名为键，DataFrame为值的字典
    """
    try:
        # 使用pandas读取所有sheet
        excel_data = pd.read_excel(file_path, sheet_name=None)
        return excel_data
    except Exception as e:
        print(f"读取Excel文件时出错: {e}")
        return None

def extract_columns(file_path, sheet_name, columns, header=0):
    """
    提取Excel指定sheet中的特定列
    
    参数:
        file_path (str): Excel文件路径
        sheet_name (str): 工作表名称
        columns (list): 需要提取的列名列表或列索引列表
        header (int): 表头行号，默认为0
    
    返回:
        DataFrame: 包含指定列的数据
    """
    try:
        df = pd.read_excel(file_path, sheet_name=sheet_name, header=header)
        return df[columns] if isinstance(columns, list) else df[[columns]]
    except Exception as e:
        print(f"Error occurred: {str(e)}")
        return None

def split_event_column(df, column_name):
    """
    将格式为"[时间戳] 事件编号：事件内容"的列拆分成三列
    参数:
        df: 原始DataFrame
        column_name: 需要拆分的列名
    返回:
        处理后的DataFrame
    """
    # 定义正则表达式匹配模式（捕获组）
    pattern = r'\[(.*?)\]\s*(\d+):\s*(.*)'
    
    # 创建临时副本避免SettingWithCopyWarning
    result_df = df.copy()
    
    # 拆分列并赋值
    result_df[['TimeStamp', 'EventType', 'EventContent']] = result_df[column_name].str.extract(pattern)
    
    # 删除原始列
    result_df.drop(column_name, axis=1, inplace=True)
    
    return result_df

def merge_tables(table1, table2, insertAfterColumn):
    # 左连接合并

    # 右表转换
    table2['Log id no'] = table2['Log id no'].astype(str)

    mergedTable = pd.merge(
        left = table1,
        right = table2[['Log id no', 'Grouping', 'Category']],
        left_on = 'EventType',
        right_on = 'Log id no',
        how='left'
    )

    # 调整列顺序
    cols = list(table1.columns)
    insert_pos = cols.index(insertAfterColumn) + 1
    new_cols = cols[:insert_pos] + ['Grouping', 'Category'] + cols[insert_pos:]

    # 根据新的列顺序返回表格
    return mergedTable[new_cols]

def encodeEntireTable(df):
    #创建原始数据副本，后续操作均在副本上进行
    df = df.copy()

    # 时间类型转换
    # 转换时间格式（YYMMDD HHMMSS -> datetime64[ns]）
    df.loc[:, 'TS'] = pd.to_datetime(df['TimeStamp'],
        format='%y%m%d %H%M%S',  # 格式说明：%y两位年份 %m月 %d日 %H时 %M分 %S秒
        errors='coerce'  # 无效时间转为NaT
    ).astype('datetime64[ns]')  # 强制转为numpy的datetime64类型

    # 事件类型编码(应该考虑全部ID还是数据集中的ID - 应考虑数据集中的ID)，通常不需要再进行归一化
    le = LabelEncoder()
    df.loc[:, 'logID_encoded'] = le.fit_transform(df['EventType'])
    num = len(le.classes_)
    le = LabelEncoder()
    df.loc[:, 'grouping_encoded'] = le.fit_transform(df['Grouping'])
    le = LabelEncoder()
    df.loc[:, 'category_encoded'] = le.fit_transform(df['Category'])

    # 星期编码（无再需归一化）
    df.loc[:, 'day_sin'] = np.sin(2 * np.pi * df['TS'].dt.dayofweek / 7)
    df.loc[:, 'day_cos'] = np.cos(2 * np.pi * df['TS'].dt.dayofweek / 7)

    # 小时编码（无再需归一化）
    df.loc[:, 'hour_sin'] = np.sin(2 * np.pi * df['TS'].dt.hour / 24)
    df.loc[:, 'hour_cos'] = np.cos(2 * np.pi * df['TS'].dt.hour / 24)

    # 分钟编码（无再需归一化）
    df.loc[:, 'minute_sin'] = np.sin(2 * np.pi * df['TS'].dt.minute / 60)
    df.loc[:, 'minute_cos'] = np.cos(2 * np.pi * df['TS'].dt.minute / 60)

    return df, num

def normalize_feature(f1, f2, matrix_list):

    matrix_np = np.array(matrix_list)
    #print(matrix_np.shape)
    feature1 = matrix_np[:, :, f1-1].reshape(-1, 1)
    feature2 = matrix_np[:, :, f2-1].reshape(-1, 1)
    
    # 初始化并训练MinMaxScaler
    scaler1 = MinMaxScaler().fit(feature1)
    scaler2 = MinMaxScaler().fit(feature2)

    # 标准化并写回原矩阵
    normalized_feature1 = scaler1.transform(feature1).reshape(matrix_np.shape[0], matrix_np.shape[1])
    matrix_np[:, :, f1-1] = normalized_feature1   
    normalized_feature2 = scaler2.transform(feature2).reshape(matrix_np.shape[0], matrix_np.shape[1])
    matrix_np[:, :, f2-1] = normalized_feature2   
    matrix_list = matrix_np.tolist()

    return matrix_list

class LSTMTimeSeries(nn.Module):

    #定义LSTM模型（hideen_size通常为特征数量的4-8倍）
    def __init__(self, hidden_size=64, num_layers=1, num_classes=2):
        super(LSTMTimeSeries, self).__init__()
        # 输入维度为11（因为每个时间步有11个特征），隐藏层大小可调
        self.lstm = nn.LSTM(input_size=11, hidden_size=hidden_size, 
                            num_layers=num_layers, batch_first=True)
        
        #self.bn = nn.BatchNorm1d(hidden_size)  # BN层（通道数=hidden_size）
        #self.relu = nn.ReLU()
        self.drop1 = nn.Dropout(0.3) #添加dropout
        self.fc = nn.Linear(hidden_size, num_classes)  # 最终输出1个值
        
    def forward(self, x):
        # x 形状: (batch, seq_len, 11)
        out, (h_n, c_n) = self.lstm(x)  
        # out的形状: (batch, seq_len, hidden_size)
        # 取最后一个时间步的输出
        last_out = out[:, -1, :]  # (batch, hidden_size)
        dropped_out = self.drop1(last_out)  # 调用dropout
        final_out = self.fc(dropped_out)   # (batch, 1)
        return final_out
    
class LSTMTimeSeriesAttention(nn.Module):

    #定义LSTM模型（hideen_size通常为特征数量的4-8倍）
    def __init__(self, hidden_size=64, num_layers=1, num_classes=2):
        super(LSTMTimeSeriesAttention, self).__init__()
        # 输入维度为11（因为每个时间步有11个特征），隐藏层大小可调
        self.lstm = nn.LSTM(input_size=11, hidden_size=hidden_size, 
                            num_layers=num_layers, batch_first=True)
        
        self.attention_weights = nn.Linear(hidden_size, 1)

        # 新增注意力机制层
        self.attention = nn.Sequential(
            nn.Linear(hidden_size, hidden_size//2),
            nn.Tanh(),
            nn.Linear(hidden_size//2, 1)
        )

        self.drop1 = nn.Dropout(0.3) #添加dropout
        self.fc = nn.Linear(hidden_size, num_classes)  # 最终输出1个值
        
    def forward(self, x):
        # x 形状: (batch, seq_len, 11)
        out, (h_n, c_n) = self.lstm(x)  
        # out的形状: (batch, seq_len, hidden_size)
        
        # 计算注意力权重
        attn_weights = self.attention(out)  # (batch, seq_len, 1)
        attn_weights = F.softmax(attn_weights.squeeze(-1), dim=1)  # (batch, seq_len)
        # 加权求和
        context = torch.bmm(attn_weights.unsqueeze(1), out).squeeze(1)

        # 取最后一个时间步的输出
        last_out = out[:, -1, :]  # (batch, hidden_size)
        dropped_out = self.drop1(context)  # 调用dropout
        final_out = self.fc(dropped_out)   # (batch, 1)
        return final_out
    
def train_one_epoch(model, optimizer, criterion, train_loader, epoch):
    model.train() #显示调用，打开梯度计算和dropout
    total_loss = 0.0
    for batch_x, batch_y in train_loader:
            optimizer.zero_grad()
            outputs = model(batch_x)
            loss = criterion(outputs, batch_y)
            loss.backward()
            optimizer.step()
            total_loss += loss.item()
    avg_train_loss = total_loss / len(train_loader)
    print(f'Epoch {epoch+1}, Train Loss: {avg_train_loss:.4f}')
    return avg_train_loss

def validate(model, test_loader, criterion, epoch):
    model.eval()
    total_loss = 0.0
    correct = 0
    with torch.no_grad():
        for x, y in test_loader:
            outputs = model(x)
            _, predicted = torch.max(outputs.data, 1)
            loss = criterion(outputs, y)
            total_loss += loss.item()
            correct += (predicted == y).sum().item()
    avg_val_loss = total_loss / len(test_loader)
    print(f'Epoch {epoch+1}, Validation Loss: {avg_val_loss:.4f}, Accuracy: {100*correct/len(test_dataset):.2f}%')
    return avg_val_loss

def datasetShuffle(train_x, test_x, train_y, test_y, isShuffle=False):
    if isShuffle:
        #将训练集和测试集先合并、打乱再重新分割 - 全局打乱
        # 合并数据集
        combined_x = np.concatenate((train_x, test_x), axis=0)
        combined_y = np.concatenate((train_y, test_y), axis=0)

        # 生成随机索引并打乱数据
        indices = np.arange(combined_x.shape[0])
        np.random.shuffle(indices)  # 原地打乱

        shuffled_x = combined_x[indices]
        shuffled_y = combined_y[indices]

        # 按原始比例分割
        train_size = np.array(train_x).shape[0]  # 原始训练集大小

        new_train_x = shuffled_x[:train_size]
        new_train_y = shuffled_y[:train_size]
        new_test_x = shuffled_x[train_size:]
        new_test_y = shuffled_y[train_size:]

        # 重新验证形状用new_否则用原始数据，而后仅在epoch内随机打乱
        print(f"New train_x shape: {new_train_x.shape}")  # 应输出 (411391, 20, 11)
        print(f"New train_y shape: {new_train_y.shape}")  # 应输出 (411391,)
        print(f"New test_x shape: {new_test_x.shape}")    # 应输出 (80770, 20, 11)
        print(f"New test_y shape: {new_test_y.shape}")    # 应输出 (80770,)
    else:
        new_train_x = train_x
        new_train_y = train_y
        new_test_x = test_x
        new_test_y = test_y
    return new_train_x, new_train_y, new_test_x, new_test_y

    decay_scheduler = StepLR(optimizer, step_size=10, gamma=0.8)   # 1-40epoch每10步变为上一步80%
    #fine_scheduler = ReduceLROnPlateau(optimizer, factor=0.1, patience=2) # 最后5epoch自适应

    # Step 11. 训练模型+验证模型+学习率动态调整
    epochs = 40
    for epoch in range(epochs):
        
        # 训练步骤
        train_loss = train_one_epoch(model, optimizer, criterion, train_loader, epoch)

        # 验证步骤
        val_loss = validate(model, test_loader, criterion, epoch)

        decay_scheduler.step()   # 逐步衰减
        
        print(f'Epoch {epoch+1}: lr={optimizer.param_groups[0]["lr"]:.2e}')
        print(f"Epoch {epoch+1}: WD={optimizer.param_groups[0]['weight_decay']:.2e}")
